// firestore.rules mein ye blocks add karein (match /databases/... ke andar)
match /reports/{id} {
  allow create: if request.auth != null && request.auth.uid == request.resource.data.reporterUid;
  allow read, update, delete: if false;   // admin panel server/Admin SDK se padhega
}
match /referralCodes/{code} {
  allow read: if request.auth != null;
  allow write: if false;
}
match /deletion_requests/{uid} {
  allow read, write: if false;
}
